export type Edition = "FR" | "EUR";
export type ReaderContext = { edition: Edition; environment: "TEST" | "PROD" };

// Public domains always decide the edition. Preview hosts always use TEST.
export function resolveReader(hostname: string): ReaderContext {
  const host = hostname.toLowerCase().replace(/\.$/, "");
  if (host === "articles.lexpress.fr" || host === "pixel-perfect-render-5344.lovable.app") return { edition: "FR", environment: "PROD" };
  if (host === "articles.lexpress.eu") return { edition: "EUR", environment: "PROD" };
  return { edition: host === "fr.localhost" ? "FR" : "EUR", environment: "TEST" };
}

export function formatPublicationDate(iso: string | null | undefined, edition: Edition): string | null {
  if (!iso) return null;
  const date = new Date(iso);
  if (!Number.isFinite(date.getTime())) return null;
  return new Intl.DateTimeFormat(editions[edition].locale, { day: "numeric", month: "long", year: "numeric", timeZone: "UTC" }).format(date);
}
